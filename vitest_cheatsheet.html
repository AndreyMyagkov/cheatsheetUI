Установка и запуск

npm i -D vitest
vitest               # watch-режим
vitest run           # однократный прогон (CI)
vitest -t "название" # только совпавшие тесты
vitest run --coverage
vitest --ui          # визуальный интерфейс

Каркас теста

import { describe, it, expect } from 'vitest';

describe('корзина', () => {
  beforeAll(() => {});   // один раз перед всеми
  beforeEach(() => {});  // перед каждым it
  afterEach(() => {});   // после каждого
  afterAll(() => {});    // один раз в конце

  it('добавляет товар', () => {
    expect(1 + 1).toBe(2);
  });

  it.skip('позже');        // пропустить
  it.only('только этот');  // изолировать
  it.todo('сделать потом');
});

info it.each([...]) / test.each — параметризованные прогоны: таблица значений с %s в имени.

Матчеры expect

  • toBe(x) — строгая равность (===); toEqual — глубокая структура
  • toStrictEqual — глубоко + учёт классов и undefined-полей
  • toBeTruthy / toBeFalsy / toBeNull / toBeUndefined
  • toContain — вхождение; toContainEqual — объект в массиве
  • toHaveLength, toMatch / toMatchObject — регэксп / частичная структура
  • toBeGreaterThan / -ThanOrEqual / toBeCloseTo(x, 2) — числа
  • toThrow('текст') / toThrowError — ожидание исключения
  • expect(fn).toHaveBeen… — см. раздел про моки
  • expect.soft(x) — не прерывает тест при провале, собирает все ошибки
  • expect(actual).toMatchInlineSnapshot() — снапшот прямо в коде

Асимметричные матчеры

Для гибких проверок внутри toEqual / objectContaining:

expect(obj).toEqual({
  id: expect.any(Number),
  at: expect.any(Date),
  tags: expect.arrayContaining(['a']),
  user: expect.objectContaining({ name: 'Иван' }),
  note: expect.stringMatching(/^OK/),
  del: expect.toBeTruthy()
});

Асинхронность: базовые стили

// 1. async-функция теста
it('грузит пользователя', async () => {
  const u = await fetchUser(1);
  expect(u.name).toBe('Анна');
});

// 2. промис возвращает expect
await expect(fetchUser(-1)).rejects.toThrow('нет id');
await expect(loadConfig()).resolves.toEqual({...});

// 3. свой таймаут (мс) и ретраи
it('медленный запрос', async () => {
  ...
}, 10_000);
// .retries(2) — на describe/уровне конфига

warn Таймаут по умолчанию 5000 мс. Unhandled rejection в тесте валит тест — не глотайте ошибки молча.

Ожидание «события»: waitFor / poll

// ждём, пока условие станет истинным
await vi.waitFor(async () => {
  expect(el.textContent).toContain('Готово');
}, { timeout: 2000, interval: 50 });

// ждём значение (не assertion):
const cfg = await vi.waitUntil(() => getConfig());

// опрос значения с матчером:
await expect.poll(() => store.count,
  { timeout: 1000 }).toBe(3);

Скорость и порядок

  • describe.concurrent — тесты внутри гоняются параллельно
  • describe.sequential — строго по очереди (когда есть shared state)
  • describe.skipIf(cond) / runIf — условные блоки
  • Файлы в Vitest всегда параллельны воркерам; состояние — на каждый файл своё

Моки, шпионы, стабы

Мок-функция: vi.fn()

import { vi } from 'vitest';
const fn = vi.fn((x) => x * 2);

fn.mockReturnValueOnce(42);       // один раз
fn.mockReturnValue(0);            // всегда
fn.mockResolvedValue({ ok: true });// Promise
fn.mockRejectedValue(new Error('e'));

expect(fn).toHaveBeenCalledWith(2);
expect(fn).toHaveBeenCalledTimes(3);
expect(fn).toHaveBeenNthCalledWith(1, 1);
expect(fn).toHaveReturnedWith(4);

fn.mock.calls;      // все вызовы: [[arg],...]
fn.mock.results;    // результаты вызовов
fn.mockClear();     // стереть вызовы
fn.mockReset();     // + сброс реализации
fn.mockRestore();   // + вернуть оригинал (для spy)

Шпион: vi.spyOn

// по умолчанию — оригинал работает, вызовы пишутся
const spy = vi.spyOn(console, 'log');
doStuff();
expect(spy).toHaveBeenCalled();

// перехватить реализацию:
spy.mockImplementation(() => 'stub');
spy.mockReturnValueOnce('x');

// вернуть как было (в afterEach!):
spy.mockRestore();

info После mockRestore spy снова прозрачен; для vi.fn() restore не имеет смысла — хватает reset.

Подмена модуля: vi.mock

// vi.mock поднимается (hoisting) ВЫШЕ импортов —
// пишем его до использования:
vi.mock('./api', () => ({
  getUser: vi.fn().mockResolvedValue({ name: 'Анна' })
}));

// частичный мок — остальное реальное:
vi.mock('./api', async (importOriginal) => ({
  ...await importOriginal(),
  getUser: vi.fn()
}));

// с фабрикой и доступом к моку:
import { getUser } from './api';
vi.mocked(getUser).mockResolvedValue(...);

// поздняя подмена в конкретном тесте:
vi.doMock('./api', () => ({ ... }));
const { getUser } = await import('./api');

info Папка __mocks__/ рядом с модулем + vi.mock('./api') без фабрики — автоматический мок.

Стабы окружения и таймеров

vi.stubGlobal('fetch', vi.fn());   // подмена globalThis.fetch
vi.unstubAllGlobals();             // вернуть

vi.stubEnv('NODE_ENV', 'test');   // переменные окружения
vi.unstubAllEnvs();

vi.useFakeTimers();               // заморозить время
vi.setSystemTime(new Date('2026-01-01'));
vi.advanceTimersByTime(1000);      // продвинуть синхронно
await vi.advanceTimersByTimeAsync(1000); // + микротаски
vi.runAllTimersAsync();            // дожать все таймеры
vi.useRealTimers();                // вернуть

API и сеть: fetch под контролем

// мок глобального fetch с Response:
const fetchMock = vi.fn()
  .mockResolvedValue(new Response(
    JSON.stringify({ ok: true }),
    { status: 200, headers: {'Content-Type':'application/json'} }));
vi.stubGlobal('fetch', fetchMock);

await loadData();
expect(fetchMock).toHaveBeenCalledWith('/api/data', expect.anything());

// ошибка сети (reject):
fetchMock.mockRejectedValue(new TypeError('Failed to fetch'));
// HTTP 500:
fetchMock.mockResolvedValue(new Response('{}', { status: 500 }));

Альтернатива — MSW (перехват на уровне сети):

import { setupServer } from 'msw/node';
import { http, HttpResponse } from 'msw';

const server = setupServer(
  http.get('/api/user', () =>
    HttpResponse.json({ name: 'Анна' }))
);
beforeAll(() => server.listen());
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

Частые ошибки

  • Мок «протекает» между тестами — чистите в afterEach: vi.restoreAllMocks(), vi.unstubAllGlobals()
  • Забытый await — тест зелёный, а ассерты не выполнились
  • vi.mock после импорта модуля не сработает — hoisting поднимет его, но фабрика уже должна быть самодостаточной (не тащить внешние let)
  • Fake timers + async-код: продвигайте время асинхронно (advanceTimersByTimeAsync), иначе промисы не успеют
  • Мок axios как модуля — не забудьте мокнуть конкретный метод (get), а не весь объект

Монтирование и опции

import { mount, shallowMount } from '@vue/test-utils';

const wrapper = mount(Counter, {
  props: { initial: 5 },
  global: {
    stubs: { RouterLink: true, teleport: true },
    provide: { theme: 'dark' },
    plugins: [createTestingPinia()],
    components: { MyIcon }
  }
});
wrapper.unmount();
  • shallowMount — дочерние компоненты заменяются заглушками
  • createTestingPinia() — тестовый стор: @pinia/testing
  • Стор с реальной логикой: setActivePinia(createPinia())

Поиск и значения

wrapper.find('[data-test="btn"]');   // нет элемента → вернёт null
wrapper.get('input');                // кинет ошибку, если нет
wrapper.findAll('.item');            // массив

await input.setValue('привет');      // v-model / input
await wrapper.setProps({ title: 'x' });
await btn.trigger('click');
await btn.trigger('submit');         // .prevent не нужен: событие на форме

warn После trigger/setValue всегда await: DOM и обновления Vue асинхронны. Для nextTick-подобного ожидания — flushPromises().

React · Testing Library

Рендер и запросы

import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';

const user = userEvent.setup();
const { rerender, unmount } = render(<Form />);

// приоритет запросов (по доступности a11y):
screen.getByRole('button', { name: /отправить/i });
screen.getByLabelText('Email');
screen.getByText('Готово');

// нет элемента?  getBy → ошибка | queryBy → null
expect(screen.queryByRole('alert')).toBeNull();
// появление позже (async):
await screen.findByRole('dialog');
await screen.findByText('Загружено');

info getBy* — элемент обязан быть; queryBy* — проверяем отсутствие; findBy* — ждём асинхронно (внутри waitFor).

Взаимодействие: userEvent vs fireEvent

// userEvent — реалистичная модель (рекомендуется):
await user.click(screen.getByRole('button'));
await user.type(screen.getByLabelText('Имя'), 'Иван');
await user.selectOptions(getByRole('combobox'), '2');

// fireEvent — низкий уровень, точечно:
import { fireEvent } from '@testing-library/react';
fireEvent.change(input, { target: { value: 'x' } });

Подводные камни

  • act-предупреждения: апдейты после await — оборачивайте в await waitFor(...)
  • Не тестируйте внутренности (state) — тестируйте то, что видит пользователь
  • vi.clearAllMocks() в afterEach — иначе вызовы копятся между тестами
  • userEvent.setup() один раз в тесте, а не в beforeEach без переустановки
